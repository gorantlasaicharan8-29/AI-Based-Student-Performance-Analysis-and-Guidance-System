import urllib.request
import json

def test_student_prediction():
    # 1. Login student
    login_url = "http://127.0.0.1:5000/api/auth/login"
    login_data = json.dumps({
        "email": "gorantlasaicharan@gmail.com",
        "password": "sai1234"
    }).encode("utf-8")
    
    req = urllib.request.Request(
        login_url,
        data=login_data,
        headers={"Content-Type": "application/json"},
        method="POST"
    )
    
    try:
        with urllib.request.urlopen(req) as res:
            login_resp = json.loads(res.read().decode())
            token = login_resp["token"]
            user_name = login_resp["user"]["name"]
            print(f"Login success: {user_name}")
    except Exception as e:
        print(f"Login failed: {e}")
        return

    # 2. Query Prediction Endpoint
    predict_url = "http://127.0.0.1:5000/api/student/predict"
    req_pred = urllib.request.Request(
        predict_url,
        headers={
            "Content-Type": "application/json",
            "Authorization": f"Bearer {token}"
        },
        method="POST"
    )
    
    try:
        with urllib.request.urlopen(req_pred) as res_pred:
            pred_resp = json.loads(res_pred.read().decode())
            print("\nAI PERFORMANCE PREDICTION OUTPUT:")
            print(json.dumps(pred_resp, indent=2))
    except Exception as e:
        print(f"Prediction query failed: {e}")

if __name__ == "__main__":
    test_student_prediction()
